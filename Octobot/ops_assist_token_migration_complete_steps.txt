import asyncio
import base64
import json
import time
from typing import Any, Optional

import httpx
from coin_auth.token import generate_m2m_token

from app.core.config import settings
from app.core.exceptions import TokenFetchError
from app.core.logging import get_logger
from app.utils.retry import async_retry


logger = get_logger(__name__)

_REFRESH_BUFFER_SECONDS = 60


def _read_jwt_exp(token: str) -> Optional[float]:
    """
    Read exp from JWT only for cache expiration calculation.
    This does NOT validate or authenticate the JWT.
    """
    try:
        parts = token.split(".")

        if len(parts) != 3:
            return None

        payload_part = parts[1]
        payload_part += "=" * (-len(payload_part) % 4)

        decoded = base64.urlsafe_b64decode(
            payload_part.encode("utf-8")
        ).decode("utf-8")

        payload = json.loads(decoded)

        exp = payload.get("exp")

        if exp is None:
            return None

        return float(exp)

    except Exception:
        return None


class TokenClient:
    """
    COIN M2M token client.

    Backend and feedback use the same:
      TOKEN_AUTH_URL
      TOKEN_CLIENT_ID
      TOKEN_CLIENT_SECRET

    Only scope is different.
    """

    def __init__(
        self,
        scope: str,
        client_name: str,
    ) -> None:
        self._scope = scope
        self._client_name = client_name
        self._token: Optional[str] = None
        self._expires_at: float = 0.0

    async def get_token(self) -> str:
        if self._token and time.time() < self._expires_at:
            return self._token

        logger.info(
            "Token expired or missing, fetching new token",
            token_client=self._client_name,
        )

        return await self._fetch_token()

    @async_retry(
        attempts=settings.HTTP_RETRY_ATTEMPTS,
        backoff_seconds=settings.HTTP_RETRY_BACKOFF_SECONDS,
        exception_types=(TokenFetchError, httpx.HTTPError),
    )
    async def _fetch_token(self) -> str:
        auth_url = settings.TOKEN_AUTH_URL
        client_id = settings.TOKEN_CLIENT_ID
        client_secret = settings.TOKEN_CLIENT_SECRET
        scope = self._scope
        verify = settings.TOKEN_VERIFY_SSL

        if not auth_url:
            raise TokenFetchError("TOKEN_AUTH_URL is not configured")

        if not client_id:
            raise TokenFetchError("TOKEN_CLIENT_ID is not configured")

        if not client_secret:
            raise TokenFetchError("TOKEN_CLIENT_SECRET is not configured")

        if not scope:
            raise TokenFetchError(
                f"Token scope is not configured for {self._client_name}"
            )

        logger.info(
            "Generating COIN M2M token",
            token_client=self._client_name,
            auth_url=auth_url,
            client_id=client_id,
            scope=scope,
            client_secret_present=True,
            client_secret_length=len(client_secret),
        )

        try:
            result: Any = await asyncio.to_thread(
                generate_m2m_token,
                auth_url,
                client_id,
                client_secret,
                scope,
                verify,
            )

        except Exception as exc:
            logger.error(
                "COIN token generation failed",
                token_client=self._client_name,
                error=str(exc),
            )

            raise TokenFetchError(
                f"COIN token generation failed for "
                f"{self._client_name}: {exc}"
            ) from exc

        access_token: Optional[str] = None
        expires_in: Optional[int] = None

        if isinstance(result, dict):
            access_token = (
                result.get("access_token")
                or result.get("accessToken")
                or result.get("token")
            )

            raw_expires_in = result.get("expires_in")

            if raw_expires_in is not None:
                try:
                    expires_in = int(raw_expires_in)
                except (TypeError, ValueError):
                    expires_in = None

        elif isinstance(result, str):
            access_token = result

        if not access_token:
            raise TokenFetchError(
                f"COIN token response for {self._client_name} "
                "did not contain an access token"
            )

        self._token = access_token

        if expires_in and expires_in > 0:
            refresh_after = max(
                expires_in - _REFRESH_BUFFER_SECONDS,
                1,
            )

            self._expires_at = time.time() + refresh_after

        else:
            jwt_exp = _read_jwt_exp(access_token)

            if jwt_exp:
                self._expires_at = max(
                    jwt_exp - _REFRESH_BUFFER_SECONDS,
                    time.time() + 1,
                )
            else:
                fallback_ttl = max(
                    settings.TOKEN_DEFAULT_TTL_SECONDS
                    - _REFRESH_BUFFER_SECONDS,
                    1,
                )

                self._expires_at = time.time() + fallback_ttl

        logger.info(
            "COIN token fetched successfully",
            token_client=self._client_name,
            expires_at=self._expires_at,
        )

        return self._token


# Backend token client
token_client = TokenClient(

import time
from unittest.mock import AsyncMock, patch

import pytest

from app.clients.token_client import TokenClient
from app.core.exceptions import TokenFetchError


# ---------------------------------------------------------------------------
# Common helper
# ---------------------------------------------------------------------------

def _make_client() -> TokenClient:
    """
    Create a TokenClient for unit tests.

    client_name="test" is intentionally kept.
    It is only used for logging/debugging and is NOT sent to COIN.
    """
    return TokenClient(
        scope="test-scope",
        client_name="test",
    )


# ---------------------------------------------------------------------------
# CACHE TEST
# ---------------------------------------------------------------------------

@pytest.mark.asyncio
async def test_get_token_uses_cache_without_fetch():
    client = _make_client()

    client._token = "cached-token"
    client._expires_at = time.time() + 120

    with patch.object(
        client,
        "_fetch_token",
        new_callable=AsyncMock,
    ) as mock_fetch:

        token = await client.get_token()

    assert token == "cached-token"
    mock_fetch.assert_not_awaited()


# ---------------------------------------------------------------------------
# SUCCESSFUL TOKEN GENERATION
# ---------------------------------------------------------------------------

@pytest.mark.asyncio
async def test_fetch_token_success():
    client = _make_client()

    generated_token = {
        "access_token": "generated-access-token",
        "expires_in": 300,
    }

    with (
        patch(
            "app.clients.token_client.generate_m2m_token",
            return_value=generated_token,
        ) as mock_generate,
        patch("app.clients.token_client.settings") as mock_settings,
    ):
        mock_settings.TOKEN_AUTH_URL = "https://token"
        mock_settings.TOKEN_CLIENT_ID = "test-client-id"
        mock_settings.TOKEN_CLIENT_SECRET = "test-secret"
        mock_settings.TOKEN_VERIFY_SSL = False
        mock_settings.TOKEN_TIMEOUT_SECONDS = 300

        token = await client._fetch_token()

    assert token == "generated-access-token"
    assert client._token == "generated-access-token"
    assert client._expires_at > time.time()

    mock_generate.assert_called_once_with(
        "https://token",
        "test-client-id",
        "test-secret",
        "test-scope",
        False,
    )

        "access_token": "abc",
        "expires_in": 300,
    }

        patch(

            return_value=generated_token,
        ) as mock_generate,

    ):

    mock_fetch.assert_awaited_once()

    backend_client = TokenClient(
        scope="backend-scope",
        client_name="backend",
    )

    feedback_client = TokenClient(
        scope="feedback-scope",
        client_name="feedback",
    )

    assert backend_client.scope == "backend-scope"
    assert backend_client.client_name == "backend"

    assert feedback_client.scope == "feedback-scope"
    assert feedback_client.client_name == "feedback"
